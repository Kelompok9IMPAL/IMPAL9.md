CREATE TABLE users (
    id_user INT PRIMARY KEY AUTO_INCREMENT,
    nama VARCHAR(100) NOT NULL,
    email VARCHAR(100) UNIQUE NOT NULL,
    password VARCHAR(255) NOT NULL,
    role ENUM('staff', 'manajer') NOT NULL,
    tim VARCHAR(50)
);

CREATE TABLE targets (
    id_target INT PRIMARY KEY AUTO_INCREMENT,
    id_user INT,
    bulan INT CHECK (bulan BETWEEN 1 AND 12),
    tahun INT,
    target_leads INT DEFAULT 0,
    target_closing INT DEFAULT 0,
    target_revenue DECIMAL(15, 2) DEFAULT 0.00,
    target_roas DECIMAL(5, 2) DEFAULT 0.00,
    FOREIGN KEY (id_user) REFERENCES users(id_user) ON DELETE CASCADE
);

CREATE TABLE daily_logs (
    id_log INT PRIMARY KEY AUTO_INCREMENT,
    id_user INT,
    tanggal DATE NOT NULL,
    jml_leads INT DEFAULT 0,
    jml_closing INT DEFAULT 0,
    nominal_revenue DECIMAL(15, 2) DEFAULT 0.00,
    nominal_spend DECIMAL(15, 2) DEFAULT 0.00,
    FOREIGN KEY (id_user) REFERENCES users(id_user) ON DELETE CASCADE
);

CREATE TABLE kpi_results (
    id_kpi INT PRIMARY KEY AUTO_INCREMENT,
    id_user INT,
    bulan INT,
    tahun INT,
    total_leads INT,
    total_closing INT,
    total_revenue DECIMAL(15, 2),
    avg_roas DECIMAL(5, 2),
    perf_score DECIMAL(5, 2),
    FOREIGN KEY (id_user) REFERENCES users(id_user) ON DELETE CASCADE
);

CREATE TABLE evaluations (
    id_evaluasi INT PRIMARY KEY AUTO_INCREMENT,
    id_kpi INT,
    id_manajer INT,
    catatan_evaluasi TEXT,
    tanggal_input TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (id_kpi) REFERENCES kpi_results(id_kpi) ON DELETE CASCADE,
    FOREIGN KEY (id_manajer) REFERENCES users(id_user)
);
